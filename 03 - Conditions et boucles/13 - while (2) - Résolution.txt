// 2
int i = 0, j = 0;
while (i <= 5)
   i += j++;
cout << i << ' ' << j;

Tour 1 : i = 0, j = 1
Tour 2 : i = 1, j = 2
Tour 3 : i = (1 + 2) = 3, j = 3
Tour 4 : i = (3 + 3) = 6, j = 4
Tour 5 : Condition false -> Sortie
i = 6, j = 4

// 3
int i = 0, j = 0;
while (i <= 5)
   i += ++j;
cout << i << ' ' << j;

Tour 0 : i = 0, j = 0
Tour 1 : i = 1, j = 1
Tour 2 : i = 3, j = 2
Tour 3 : i = 6, j = 3
Tour 4 : Condition false -> Sortie
i = 6, j = 3

// 4
int i = 0, j = 0;
while (j <= 5)
   i += j++;
cout << i << ' ' << j;

Tour 0 : i = 0              / j = 0
Tour 1 : i = 0              / j = 1
Tour 2 : i = 1              / j = 2
Tour 3 : i = (1 + 2) = 3    / j = 3
Tour 4 : i = (3 + 3) = 6    / j = 4
Tour 5 : i = (6 + 4) = 10   / j = 5
Tour 6 : i = (10 + 5) = 15  / j = 6
Tour 7 : Sortie de boucle
i = 15 et j = 6

//5

Tour 0 : i = 0              / j = 0
Tour 1 : i = 1              / j = 1
Tour 2 : i = (1 + 2) = 3    / j = 2     (+ 2, car ++j, donc il additionne la valeur post-incrémentée)
Tour 3 : i = (3 + 3) = 6    / j = 3
Tour 4 : i = (6 + 4) = 10   / j = 4
Tour 5 : i = (10 + 5) = 15  / j = 5
Tour 6 : i = (15 + 6) = 21  / j = 6
Tour 7 : Sortie de boucle
i = 21 et j = 6

// 6

Attention piège ici, le j++ n'est pas dans la boucle !!!!

Tour 1 : i = 2              / j = 1
Tour 2 : i = 4              / j = 1
Tour 3 : i = 6              / j = 1
